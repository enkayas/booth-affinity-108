function doGet(e) {
  var p = (e && e.parameter) ? e.parameter : {};
  var action = String(p.action || '').trim();

  var out;
  try {
    if (action === 'getDashboardSummary') {
      out = getDashboardSummary_(p);
    } else if (action === 'getSavedAffinity') {
      out = getSavedAffinity_(p);
    } else {
      out = { ok: false, message: 'Unknown action' };
    }
  } catch (err) {
    out = { ok: false, message: err && err.message ? err.message : String(err) };
  }

  return respond_(out, p.callback);
}

function doPost(e) {
  var body = {};
  try {
    body = JSON.parse(e.postData && e.postData.contents ? e.postData.contents : '{}');
  } catch (_) {}

  var action = String(body.action || '').trim();
  var out;

  try {
    if (action === 'saveBoothAffinities') {
      out = saveBoothAffinities_(body);
    } else if (action === 'clearBoothData') {
      out = clearBoothData_(body);
    } else {
      out = { ok: false, message: 'Unknown action' };
    }
  } catch (err) {
    out = { ok: false, message: err && err.message ? err.message : String(err) };
  }

  return ContentService
    .createTextOutput(JSON.stringify(out))
    .setMimeType(ContentService.MimeType.JSON);
}

function getDashboardSummary_(params) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();

  var summarySheetName =
    PropertiesService.getScriptProperties().getProperty('DASHBOARD_SUMMARY_SHEET_NAME') ||
    'Booth Affinity Master';

  var sh = ss.getSheetByName(summarySheetName);
  if (!sh) {
    return { ok: false, message: 'Dashboard summary sheet not found: ' + summarySheetName };
  }

  var rows = readSheetObjects_(sh);
  var mandalFilter = cleanText_(params && params.mandal);
  var villageFilter = cleanText_(params && params.village);

  var booths = rows
    .map(function(row) {
      var boothNo = toNumber_(pick_(row, ['booth', 'booth no', 'booth number', 'booth#', 'booth no.']));
      if (!boothNo) return null;

      var mandal = cleanText_(pick_(row, ['mandal']));
      var village = cleanText_(pick_(row, ['village', 'village/town', 'town', 'village town']));
      var totalVoters = toNumber_(pick_(row, ['totalvoters', 'total voters', 'total_voters']));

      var summary = {
        A: toNumber_(pick_(row, ['a'])),
        B: toNumber_(pick_(row, ['b'])),
        C: toNumber_(pick_(row, ['c'])),
        D: toNumber_(pick_(row, ['d'])),
        E: toNumber_(pick_(row, ['e']))
      };

      var completed = summary.A + summary.B + summary.C + summary.D + summary.E;
      var completionPct = totalVoters ? Math.round((completed / totalVoters) * 100) : 0;

      return {
        booth: boothNo,
        mandal: mandal,
        village: village,
        totalVoters: totalVoters,
        completed: completed,
        completionPct: completionPct,
        summary: summary
      };
    })
    .filter(Boolean)
    .filter(function(item) {
      var mandalOk = !mandalFilter || mandalFilter === '__all__' || item.mandal === mandalFilter;
      var villageOk = !villageFilter || villageFilter === '__all__' || item.village === villageFilter;
      return mandalOk && villageOk;
    })
    .sort(function(a, b) {
      return a.booth - b.booth;
    });

  return { ok: true, booths: booths };
}

function getSavedAffinity_(params) {
  var boothNo = toNumber_(params && params.booth);
  if (!boothNo) return { ok: false, message: 'Invalid booth' };

  var sheet = getAffinitySheet_();
  var values = sheet.getDataRange().getValues();
  if (!values || values.length < 2) return { ok: true, saved: [] };

  var headers = values[0].map(function(h) {
    return normalizeHeader_(h);
  });

  var iBooth = findHeaderIndexByNormalized_(headers, ['booth', 'booth no', 'booth number', 'booth#', 'booth no.']);
  var iSlNo = findHeaderIndexByNormalized_(headers, ['slno', 'sl no', 'sl_no', 'serial no', 'serial_no']);
  var iAffinity = findHeaderIndexByNormalized_(headers, ['affinity', 'value']);
  var iRelocated = findHeaderIndexByNormalized_(headers, ['relocated']);
  var iVoted = findHeaderIndexByNormalized_(headers, ['voted']);
  var iUpdatedAt = findHeaderIndexByNormalized_(headers, ['updatedat', 'updated at']);

  if (iBooth < 0 || iSlNo < 0 || iAffinity < 0) {
    return { ok: false, message: 'Affinity sheet headers missing required columns' };
  }

  var bySl = {};
  var timeBySl = {};

  for (var r = 1; r < values.length; r++) {
    var row = values[r];
    if (toNumber_(row[iBooth]) !== boothNo) continue;

    var slNo = toNumber_(row[iSlNo]);
    var affinity = String(row[iAffinity] == null ? '' : row[iAffinity]).trim().toUpperCase();

    if (!slNo) continue;
    if (!/^[A-E]$/.test(affinity)) continue;

    var ts = r;
    if (iUpdatedAt >= 0) {
      var d = new Date(row[iUpdatedAt]);
      if (!isNaN(d.getTime())) ts = d.getTime();
    }

    if (timeBySl[slNo] === undefined || ts >= timeBySl[slNo]) {
      timeBySl[slNo] = ts;
      bySl[slNo] = {
        affinity: affinity,
        relocated: readBooleanFlag_(iRelocated >= 0 ? row[iRelocated] : false),
        voted: readBooleanFlag_(iVoted >= 0 ? row[iVoted] : false)
      };
    }
  }

  var out = Object.keys(bySl)
    .map(function(k) {
      return {
        slNo: Number(k),
        affinity: bySl[k].affinity,
        relocated: bySl[k].relocated,
        voted: bySl[k].voted
      };
    })
    .sort(function(a, b) {
      return a.slNo - b.slNo;
    });

  return { ok: true, saved: out };
}

function saveBoothAffinities_(body) {
  var boothNo = toNumber_(body && body.booth);
  var voters = body && Array.isArray(body.voters) ? body.voters : null;

  if (!boothNo) return { ok: false, message: 'Invalid booth' };
  if (!voters) return { ok: false, message: 'Invalid voters payload' };

  var lock = LockService.getScriptLock();
  lock.waitLock(30000);

  try {
    var sheet = getAffinitySheet_();
    ensureAffinityHeaders_(sheet);

    var data = sheet.getDataRange().getValues();
    var headers = data[0].map(function(h) {
      return normalizeHeader_(h);
    });

    var iBooth = findHeaderIndexByNormalized_(headers, ['booth', 'booth no', 'booth number', 'booth#', 'booth no.']);
    var iSlNo = findHeaderIndexByNormalized_(headers, ['slno', 'sl no', 'sl_no', 'serial no', 'serial_no']);
    var iAffinity = findHeaderIndexByNormalized_(headers, ['affinity', 'value']);
    var iRelocated = findHeaderIndexByNormalized_(headers, ['relocated']);
    var iVoted = findHeaderIndexByNormalized_(headers, ['voted']);
    var iUpdatedAt = findHeaderIndexByNormalized_(headers, ['updatedat', 'updated at']);

    if (iBooth < 0 || iSlNo < 0 || iAffinity < 0) {
      throw new Error('Affinity sheet headers missing required columns');
    }

    var incoming = {};
    voters.forEach(function(v) {
      var slNo = toNumber_(v && v.slNo);
      var affinity = String(v && v.affinity ? v.affinity : '').trim().toUpperCase();
      if (!slNo) return;
      incoming[slNo] = {
        affinity: /^[A-E]$/.test(affinity) ? affinity : '',
        relocated: !!v.relocated,
        voted: !!v.voted
      };
    });

    var retained = [data[0]];
    for (var r = 1; r < data.length; r++) {
      if (toNumber_(data[r][iBooth]) !== boothNo) {
        retained.push(data[r]);
      }
    }

    var summary = { A: 0, B: 0, C: 0, D: 0, E: 0 };
    var boothRows = [];
    var now = new Date();
    var width = data[0].length;
    var relocatedCount = 0;
    var votedCount = 0;

    Object.keys(incoming)
      .map(function(k) {
        return Number(k);
      })
      .sort(function(a, b) {
        return a - b;
      })
      .forEach(function(slNo) {
        var incoming_row = incoming[slNo];
        var affinity = incoming_row.affinity;
        if (!/^[A-E]$/.test(affinity)) return;

        var row = [];
        for (var c = 0; c < width; c++) row.push('');

        row[iBooth] = boothNo;
        row[iSlNo] = slNo;
        row[iAffinity] = affinity;
        if (iRelocated >= 0) row[iRelocated] = incoming_row.relocated ? 'TRUE' : '';
        if (iVoted >= 0) row[iVoted] = incoming_row.voted ? 'TRUE' : '';
        if (iUpdatedAt >= 0) row[iUpdatedAt] = now;

        boothRows.push(row);
        summary[affinity]++;
        if (incoming_row.relocated) relocatedCount++;
        if (incoming_row.voted) votedCount++;
      });

    var allRows = retained.concat(boothRows);

    sheet.clearContents();
    if (allRows.length > 0) {
      sheet.getRange(1, 1, allRows.length, width).setValues(allRows);
    }

    var completed = summary.A + summary.B + summary.C + summary.D + summary.E;
    var completionPct = upsertBoothMasterSummary_(boothNo, summary, completed, relocatedCount, votedCount);

    return {
      ok: true,
      summary: summary,
      completionPct: completionPct,
      relocatedTotal: relocatedCount,
      votedTotal: votedCount
    };
  } finally {
    lock.releaseLock();
  }
}

function clearBoothData_(body) {
  var boothNo = toNumber_(body && body.booth);
  if (!boothNo) return { ok: false, message: 'Invalid booth' };

  var lock = LockService.getScriptLock();
  lock.waitLock(30000);

  try {
    var sheet = getAffinitySheet_();
    var values = sheet.getDataRange().getValues();

    if (!values || values.length < 2) {
      return { ok: true, message: 'Affinity sheet is empty' };
    }

    var headers = values[0].map(function(h) {
      return normalizeHeader_(h);
    });

    var iBooth = findHeaderIndexByNormalized_(headers, ['booth', 'booth no', 'booth number', 'booth#', 'booth no.']);

    if (iBooth < 0) {
      throw new Error('Affinity sheet headers missing booth column');
    }

    var retained = [values[0]];
    for (var r = 1; r < values.length; r++) {
      if (toNumber_(values[r][iBooth]) !== boothNo) {
        retained.push(values[r]);
      }
    }

    sheet.clearContents();
    if (retained.length > 0) {
      sheet.getRange(1, 1, retained.length, values[0].length).setValues(retained);
    }

    var summary = { A: 0, B: 0, C: 0, D: 0, E: 0 };
    var completionPct = upsertBoothMasterSummary_(boothNo, summary, 0, 0, 0);

    return {
      ok: true,
      message: 'All booth data cleared successfully',
      completionPct: completionPct
    };
  } finally {
    lock.releaseLock();
  }
}

function upsertBoothMasterSummary_(boothNo, summary, completed, relocatedTotal, votedTotal) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var summarySheetName =
    PropertiesService.getScriptProperties().getProperty('DASHBOARD_SUMMARY_SHEET_NAME') ||
    'Booth Affinity Master';

  var sh = ss.getSheetByName(summarySheetName);
  if (!sh) return 0;

  var values = sh.getDataRange().getValues();
  if (!values || values.length < 1) return 0;

  var headers = values[0].map(function(h) {
    return normalizeHeader_(h);
  });

  var iBooth = findHeaderIndexByNormalized_(headers, ['booth', 'booth no', 'booth number', 'booth#', 'booth no.']);
  var iTotal = findHeaderIndexByNormalized_(headers, ['totalvoters', 'total voters', 'total_voters']);
  var iCompleted = findHeaderIndexByNormalized_(headers, ['completed', 'completed voters', 'completed_voters', 'filled']);
  var iCompletionPct = findHeaderIndexByNormalized_(headers, ['completionpct', 'completion %', 'completion percentage', 'completion']);
  var iRelocatedTot = findHeaderIndexByNormalized_(headers, ['relocated_tot', 'relocated total', 'relocated tot']);
  var iVotedTot = findHeaderIndexByNormalized_(headers, ['voted_tot', 'voted total', 'voted tot']);
  var iA = findHeaderIndexByNormalized_(headers, ['a']);
  var iB = findHeaderIndexByNormalized_(headers, ['b']);
  var iC = findHeaderIndexByNormalized_(headers, ['c']);
  var iD = findHeaderIndexByNormalized_(headers, ['d']);
  var iE = findHeaderIndexByNormalized_(headers, ['e']);

  if (iBooth < 0) return 0;

  var targetRow = -1;
  for (var r = 1; r < values.length; r++) {
    if (toNumber_(values[r][iBooth]) === boothNo) {
      targetRow = r + 1;
      break;
    }
  }
  if (targetRow < 0) return 0;

  var totalVoters = iTotal >= 0 ? toNumber_(values[targetRow - 1][iTotal]) : 0;
  var completionPct = totalVoters ? Math.round((completed / totalVoters) * 100) : 0;

  if (iA >= 0) sh.getRange(targetRow, iA + 1).setValue(summary.A);
  if (iB >= 0) sh.getRange(targetRow, iB + 1).setValue(summary.B);
  if (iC >= 0) sh.getRange(targetRow, iC + 1).setValue(summary.C);
  if (iD >= 0) sh.getRange(targetRow, iD + 1).setValue(summary.D);
  if (iE >= 0) sh.getRange(targetRow, iE + 1).setValue(summary.E);
  if (iCompleted >= 0) sh.getRange(targetRow, iCompleted + 1).setValue(completed);
  if (iCompletionPct >= 0) sh.getRange(targetRow, iCompletionPct + 1).setValue(completionPct);
  if (iRelocatedTot >= 0) sh.getRange(targetRow, iRelocatedTot + 1).setValue(Number(relocatedTotal || 0));
  if (iVotedTot >= 0) sh.getRange(targetRow, iVotedTot + 1).setValue(Number(votedTotal || 0));

  return completionPct;
}

function getAffinitySheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var name = PropertiesService.getScriptProperties().getProperty('AFFINITY_SHEET_NAME');
  if (!name) throw new Error('Missing script property: AFFINITY_SHEET_NAME');

  var sh = ss.getSheetByName(name);
  if (!sh) throw new Error('Affinity sheet not found: ' + name);
  return sh;
}

function ensureAffinityHeaders_(sheet) {
  if (sheet.getLastRow() > 0) return;
  sheet.getRange(1, 1, 1, 6).setValues([['booth', 'slNo', 'affinity', 'relocated', 'voted', 'updatedAt']]);
}

function findHeaderIndexByNormalized_(headers, aliases) {
  for (var i = 0; i < aliases.length; i++) {
    var idx = headers.indexOf(normalizeHeader_(aliases[i]));
    if (idx >= 0) return idx;
  }
  return -1;
}

function respond_(obj, callback) {
  var json = JSON.stringify(obj || {});
  if (callback) {
    return ContentService
      .createTextOutput(callback + '(' + json + ');')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService
    .createTextOutput(json)
    .setMimeType(ContentService.MimeType.JSON);
}

function readSheetObjects_(sheet) {
  var values = sheet.getDataRange().getValues();
  if (!values || !values.length) return [];

  var headers = values[0].map(function(h) {
    return normalizeHeader_(h);
  });

  var out = [];
  for (var r = 1; r < values.length; r++) {
    var row = {};
    for (var c = 0; c < headers.length; c++) {
      row[headers[c]] = values[r][c];
    }
    out.push(row);
  }
  return out;
}

function pick_(obj, aliases) {
  if (!obj) return '';
  for (var i = 0; i < aliases.length; i++) {
    var key = normalizeHeader_(aliases[i]);
    if (Object.prototype.hasOwnProperty.call(obj, key)) {
      return obj[key];
    }
  }
  return '';
}

function toNumber_(value) {
  if (value === null || value === undefined || value === '') return 0;
  var n = Number(String(value).replace(/,/g, '').trim());
  return isNaN(n) ? 0 : n;
}

function cleanText_(value) {
  return String(value == null ? '' : value).trim().toUpperCase();
}

function normalizeHeader_(value) {
  return String(value == null ? '' : value)
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, ' ')
    .trim();
}

function readBooleanFlag_(value) {
  if (value === true) return true;
  var normalized = String(value == null ? '' : value).trim().toLowerCase();
  return normalized === 'true' || normalized === '1' || normalized === 'yes' || normalized === 'y';
}
